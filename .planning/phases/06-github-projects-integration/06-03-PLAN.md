### 6. Navigation Flow

## Implementation Details

### Backend: Import Endpoint

```python
@router.post(
    "/import",
    response_model=GitHubImportResponse,
    summary="Import GitHub repositories to resume",
)
async def import_repositories(
    request: GitHubImportRequest,
    current_user: Annotated[User, Depends(get_current_user)],
    db: Annotated[AsyncSession, Depends(get_async_session)],
):
    """
    Import selected GitHub repositories as resume projects.

    Takes repository IDs and customizations, fetches repo details,
    and returns formatted project entries for the resume.

    **Request:**
    - `repositories`: List of repositories to import with customizations
    - Each repo includes: id, owner, name, and optional customizations

    **Response:**
    - `success`: Whether import succeeded
    - `imported_count`: Number of projects imported
    - `projects`: List of formatted project entries
    """
    # Validate user has GitHub connection
    github_connection = await get_user_github_connection(current_user.id, db)
    if not github_connection:
        raise HTTPException(400, "GitHub account not connected")

    # Decrypt OAuth token
    access_token = decrypt_token(github_connection.access_token)

    # Fetch and format each repository
    imported_projects = []
    async with GitHubAPIClient(access_token) as gh_client:
        for repo_request in request.repositories:
            # Fetch repo details
            repo = await gh_client.get_repo(repo_request.owner, repo_request.name)

            # Apply customizations or use defaults
            project = format_repo_as_project(repo, repo_request.customizations)
            imported_projects.append(project)

    return GitHubImportResponse(
        success=True,
        imported_count=len(imported_projects),
        projects=imported_projects,
    )
```

### Frontend: Import Preview Component

```tsx
const GitHubImportPreview: React.FC<{
  selectedRepos: GitHubRepository[];
  onComplete: (projects: ProjectEntry[]) => void;
  onBack: () => void;
  onCancel: () => void;
}> = ({ selectedRepos, onComplete, onBack, onCancel }) => {
  const {
    importCustomizations,
    updateCustomization,
    generateProjectDetails,
    importToResume,
    isImporting,
    importError,
  } = useGitHubImportStore();

  const handleAutoGenerate = async (repoId: number) => {
    await generateProjectDetails(repoId);
  };

  const handleImport = async () => {
    try {
      await importToResume();
      // Get imported projects from store and call onComplete
      const projects = useGitHubImportStore.getState().importedProjects;
      onComplete(projects);
    } catch (error) {
      // Error handled in store
    }
  };

  return (
    <Dialog isOpen onClose={onCancel} title="Review Your Selection" size="large">
      <div className="import-preview">
        <p className="summary">
          {selectedRepos.length} repositories selected for import
        </p>

        {selectedRepos.map(repo => (
          <ProjectCustomizationForm
            key={repo.id}
            repository={repo}
            customization={importCustomizations[repo.id]}
            onUpdate={(updates) => updateCustomization(repo.id, updates)}
            onAutoGenerate={() => handleAutoGenerate(repo.id)}
          />
        ))}

        {importError && (
          <Alert variant="error">{importError}</Alert>
        )}

        <DialogActions>
          <Button onClick={onBack} variant="secondary">
            ← Back
          </Button>
          <Button
            onClick={handleImport}
            disabled={isImporting}
            variant="primary"
          >
            {isImporting ? 'Importing...' : 'Import to Resume'}
          </Button>
        </DialogActions>
      </div>
    </Dialog>
  );
};
```

          type="text"

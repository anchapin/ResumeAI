### GitHubRepoSelector Component

```tsx
const GitHubRepoSelector: React.FC<{
  onComplete: (selectedIds: number[]) => void;
  onCancel: () => void;
}> = ({ onComplete, onCancel }) => {
  const {
    repositories,
    selectedRepoIds,
    isFetchingRepos,
    reposError,
    fetchGitHubRepositories,
    toggleRepoSelection,
    selectAllRepos,
    clearRepoSelection,
    // ... other state and actions
  } = useGitHubStore();

  // Fetch repos on mount
  useEffect(() => {
    fetchGitHubRepositories({ page: 1 });
  }, []);

  // Handle search debounce
  const debouncedSearch = useDebouncedValue(searchQuery, 300);

  // Filter and sort repositories
  const filteredRepos = useMemo(() => {
    return repositories
      .filter(repo => matchesSearch(repo, searchQuery))
      .sort(getSortComparator(sortBy, sortDirection));
  }, [repositories, searchQuery, sortBy, sortDirection]);

  return (
    <Dialog isOpen onClose={onCancel} title="Import GitHub Projects">
      {/* Search and filters */}
      <GitHubRepoFilters />
      
      {/* Repository grid */}
      <div className="repo-grid">
        {filteredRepos.map(repo => (
          <GitHubRepoCard
            key={repo.id}
            repo={repo}
            isSelected={selectedRepoIds.includes(repo.id)}
            onToggle={() => toggleRepoSelection(repo.id)}
          />
        ))}
      </div>
      
      {/* Pagination */}
      <Pagination
        currentPage={currentPage}
        totalPages={Math.ceil(totalRepos / perPage)}
        onPageChange={setCurrentPage}
      />
      
      {/* Action buttons */}
      <DialogActions>
        <Button onClick={onCancel}>Cancel</Button>
        <Button
          variant="primary"
          onClick={() => onComplete(selectedRepoIds)}
          disabled={selectedRepoIds.length === 0}
        >
          Next: Review Selection ({selectedRepoIds.length})
        </Button>
      </DialogActions>
    </Dialog>
  );
};
```

### Custom Hook: useGitHubRepositories

```typescript
export function useGitHubRepositories() {
  const store = useGitHubStore();
  
  const fetchRepos = useCallback(async (page: number = 1) => {
    store.setIsFetchingRepos(true);
    store.setReposError(null);
    
    try {
      const params = {
        page,
        per_page: store.perPage,
        type: store.filterType,
        sort: store.sortBy,
        direction: store.sortDirection,
      };
      
      const response = await fetchGitHubRepositories(params);
      
      store.setRepositories(prev =>
        page === 1 ? response.repositories : [...prev, ...response.repositories]
      );
      store.setTotalRepos(response.total_count);
      store.setHasMoreRepos(response.has_more);
      store.setCurrentPage(page);
    } catch (error) {
      store.setReposError(
        error instanceof Error ? error.message : 'Failed to fetch repositories'
      );
    } finally {
      store.setIsFetchingRepos(false);
    }
  }, [store]);
  
  return {
    ...store,
    fetchRepos,
  };
}
```

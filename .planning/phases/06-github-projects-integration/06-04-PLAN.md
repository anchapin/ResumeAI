### 4. Error Handling Improvements

**Backend Error Responses:**

```python
class GitHubImportError(Exception):
    """Base error for GitHub import operations."""
    pass

class GitHubTokenExpiredError(GitHubImportError):
    """Raised when GitHub OAuth token is expired/invalid."""
    
    def to_response(self):
        return {
            "error": "token_expired",
            "message": "Your GitHub connection has expired. Please reconnect.",
            "action": "reconnect",
        }

class GitHubRateLimitError(GitHubImportError):
    """Raised when GitHub API rate limit is exceeded."""
    
    def __init__(self, retry_after: int):
        self.retry_after = retry_after
    
    def to_response(self):
        return {
            "error": "rate_limit",
            "message": "GitHub API rate limit exceeded.",
            "retry_after": self.retry_after,
            "retry_at": datetime.now() + timedelta(seconds=self.retry_after),
        }

class GitHubRepoNotFoundError(GitHubImportError):
    """Raised when a repository is not found."""
    pass
```

**Frontend Error Handling:**

```typescript
// utils/githubErrorHandler.ts

export function handleGitHubError(error: unknown): ErrorAction {
  if (error instanceof GitHubAPIError) {
    switch (error.status_code) {
      case 401:
        return {
          type: 'reconnect',
          title: 'GitHub Connection Expired',
          message: 'Your GitHub connection has expired. Please reconnect to continue.',
          action: 'reconnect',
        };
      case 403:
        if (error.response_data?.retry_after) {
          return {
            type: 'rate_limit',
            title: 'Rate Limit Exceeded',
            message: `Too many requests. Please wait ${error.response_data.retry_after} seconds.`,
            retryAfter: error.response_data.retry_after,
          };
        }
        return {
          type: 'forbidden',
          title: 'Access Denied',
          message: 'You do not have access to this repository.',
        };
      case 404:
        return {
          type: 'not_found',
          title: 'Repository Not Found',
          message: 'The requested repository could not be found.',
        };
      default:
        return {
          type: 'unknown',
          title: 'GitHub Error',
          message: error.message || 'An unexpected error occurred.',
        };
    }
  }
  
  // Network errors
  if (error instanceof TypeError && error.message.includes('fetch')) {
    return {
      type: 'network',
      title: 'Network Error',
      message: 'Unable to connect to GitHub. Please check your internet connection.',
      action: 'retry',
    };
  }
  
  // Default error
  return {
    type: 'unknown',
    title: 'Error',
    message: 'An unexpected error occurred. Please try again.',
    action: 'retry',
  };
}
```

### 5. Caching Implementation

**Backend Caching:**

```python
# lib/cache/github_cache.py

import asyncio
from typing import Dict, Any, Optional
from datetime import datetime, timedelta

class GitHubCache:
    """In-memory cache for GitHub API responses."""
    
    def __init__(self):
        self._cache: Dict[str, Dict[str, Any]] = {}
        self._locks: Dict[str, asyncio.Lock] = {}
    
    def _get_key(self, user_id: int, endpoint: str, params: Dict) -> str:
        """Generate cache key."""
        params_str = '&'.join(f'{k}={v}' for k, v in sorted(params.items()))
        return f"github:{user_id}:{endpoint}?{params_str}"
    
    async def get(self, key: str) -> Optional[Dict[str, Any]]:
        """Get value from cache if not expired."""
        if key not in self._cache:
            return None
        
        entry = self._cache[key]
        if datetime.now() > entry['expires_at']:
            del self._cache[key]
            return None
        
        return entry['data']
    
    async def set(
        self,
        key: str,
        data: Dict[str, Any],
        ttl_seconds: int = 300,
    ) -> None:
        """Set value in cache with TTL."""
        self._cache[key] = {
            'data': data,
            'expires_at': datetime.now() + timedelta(seconds=ttl_seconds),
        }
    
    async def invalidate(self, user_id: int, pattern: str = None) -> None:
        """Invalidate cache entries for a user."""
        prefix = f"github:{user_id}:"
        keys_to_delete = [
            key for key in self._cache.keys()
            if key.startswith(prefix) and (pattern is None or pattern in key)
        ]
        for key in keys_to_delete:
            del self._cache[key]

# Global cache instance
github_cache = GitHubCache()

# Cache decorators
def cache_github_repos(ttl: int = 300):
    """Decorator to cache repository list responses."""
    def decorator(func):
        async def wrapper(user_id: int, *args, **kwargs):
            key = f"github:{user_id}:repos:{args}:{kwargs}"
            
            cached = await github_cache.get(key)
            if cached:
                return cached
            
            result = await func(user_id, *args, **kwargs)
            await github_cache.set(key, result, ttl)
            return result
        return wrapper
    return decorator
```

**Frontend Caching:**

```typescript
// utils/githubCache.ts

interface CacheEntry<T> {
  data: T;
  timestamp: number;
  expiresAt: number;
}

class GitHubCache<T> {
  private cache: Map<string, CacheEntry<T>> = new Map();
  private ttl: number;

  constructor(ttlSeconds: number = 300) {
    this.ttl = ttlSeconds * 1000; // Convert to ms
  }

  private getKey(key: string): string {
    return `github:${key}`;
  }

  get(key: string): T | null {
    const entry = this.cache.get(this.getKey(key));
    if (!entry) return null;

    if (Date.now() > entry.expiresAt) {
      this.cache.delete(this.getKey(key));
      return null;
    }

    return entry.data;
  }

  set(key: string, data: T): void {
    const now = Date.now();
    this.cache.set(this.getKey(key), {
      data,
      timestamp: now,
      expiresAt: now + this.ttl,
    });
  }

  invalidate(pattern?: string): void {
    if (!pattern) {
      this.cache.clear();
      return;
    }

    const prefixedPattern = this.getKey(pattern);
    for (const key of this.cache.keys()) {
      if (key.startsWith(prefixedPattern)) {
        this.cache.delete(key);
      }
    }
  }
}

// Usage in API client
const reposCache = new GitHubCache<GitHubRepoListResponse>(300);

export async function fetchGitHubRepositories(
  params: GitHubRepoListParams
): Promise<GitHubRepoListResponse> {
  const cacheKey = `repos:${JSON.stringify(params)}`;
  
  // Try cache first
  const cached = reposCache.get(cacheKey);
  if (cached) {
    return cached;
  }

  // Fetch from API
  const response = await apiCall(params);
  
  // Cache the result
  reposCache.set(cacheKey, response);
  
  return response;
}
```

```typescript
// hooks/useDebouncedValue.ts
export function useDebouncedValue<T>(value: T, delay: number): T {
  const [debouncedValue, setDebouncedValue] = useState<T>(value);

  useEffect(() => {
    const timer = setTimeout(() => setDebouncedValue(value), delay);
    return () => clearTimeout(timer);
  }, [value, delay]);

  return debouncedValue;
}
```

```python
# Concurrent fetching with limits
import asyncio

async def fetch_multiple_repos(
    client: GitHubAPIClient,
    repo_requests: List[RepoRequest],
    max_concurrent: int = 5,
) -> List[GitHubRepository]:
    """Fetch multiple repos concurrently with limit."""
    semaphore = asyncio.Semaphore(max_concurrent)
    
    async def fetch_with_semaphore(req):
        async with semaphore:
            return await client.get_repo(req.owner, req.name)
    
    tasks = [fetch_with_semaphore(req) for req in repo_requests]
    results = await asyncio.gather(*tasks, return_exceptions=True)
    
    # Filter out exceptions
    return [r for r in results if isinstance(r, GitHubRepository)]
```
